import pandas as pd
import numpy as np
import json
import matplotlib.pyplot as plt
import seaborn as sns

# A1. Load CSV and display first five rows
df = pd.read_csv("urban_mobility.csv")
print(df.head())

# A2. Create Energy_Per_Km
df["Energy_Per_Km"] = df["Energy_Used_kWh"] / df["Distance_Km"]
print(df.head())

# A3. Electric Van records with Maintenance_Flag = Yes
electric_vans = df[
    (df["Vehicle_Type"] == "Electric Van") &
    (df["Maintenance_Flag"] == "Yes")
]
print(electric_vans)

# A4. Group by Vehicle_Type
grouped = df.groupby("Vehicle_Type").agg({
    "Distance_Km": "mean",
    "Trips_Count": "sum"
})
print(grouped)

# A5. NumPy statistics
energy = df["Energy_Used_kWh"].to_numpy()

print("Mean:", np.mean(energy))
print("Max :", np.max(energy))
print("Min :", np.min(energy))

# A6. Bar chart
avg_energy = df.groupby("Vehicle_Type")["Energy_Used_kWh"].mean()

plt.figure(figsize=(8,5))
sns.barplot(x=avg_energy.index, y=avg_energy.values)
plt.title("Average Energy Used by Vehicle Type")
plt.xlabel("Vehicle Type")
plt.ylabel("Average Energy Used (kWh)")
plt.xticks(rotation=45)
plt.show()

CREATE OR REPLACE VIEW purchase_order_summary AS
SELECT
    po.po_id,
    s.supplier_name,
    p.part_name,
    poi.quantity,
    p.unit_price,
    poi.quantity * p.unit_price AS line_total,
    po.status
FROM purchase_orders po
JOIN suppliers s
ON po.supplier_id = s.supplier_id
JOIN purchase_order_items poi
ON po.po_id = poi.po_id
JOIN parts p
ON poi.part_id = p.part_id;

SELECT *
FROM purchase_order_summary
WHERE status = 'Pending';


-- C11
CREATE OR REPLACE FUNCTION calculate_part_discount(amount NUMERIC)
RETURNS NUMERIC
LANGUAGE plpgsql
AS
$$
BEGIN
    IF amount >= 2000 THEN
        RETURN amount * 0.10;
    ELSE
        RETURN 0;
    END IF;
END;
$$;

SELECT part_name,
       unit_price,
       calculate_part_discount(unit_price) AS discount
FROM parts;


-- C12
CREATE OR REPLACE FUNCTION final_part_price(amount NUMERIC)
RETURNS NUMERIC
LANGUAGE plpgsql
AS
$$
BEGIN
    RETURN amount - calculate_part_discount(amount);
END;
$$;

SELECT part_name,
       unit_price,
       final_part_price(unit_price) AS final_price
FROM parts;

CREATE OR REPLACE PROCEDURE update_po_status(
    p_po_id INT,
    p_new_status VARCHAR
)
LANGUAGE plpgsql
AS
$$
BEGIN
    UPDATE purchase_orders
    SET status = p_new_status
    WHERE po_id = p_po_id;
END;
$$;

CALL update_po_status(1, 'Received');

SELECT *
FROM purchase_orders
WHERE po_id = 1;


-- C14
CREATE OR REPLACE PROCEDURE mark_old_pending_orders()
LANGUAGE plpgsql
AS
$$
BEGIN
    UPDATE purchase_orders
    SET status = 'Received'
    WHERE status = 'Pending'
      AND order_date < DATE '2026-10-04';
END;
$$;

CALL mark_old_pending_orders();

SELECT *
FROM purchase_orders;
